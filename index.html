// PipNex Flow - live EURUSD feed const PIPNEX_API = "https://pipnix-flow-api.onrender.com"; async function loadEURUSD() { try { const response = await fetch( `${PIPNEX_API}/api/market/forex/EURUSD` ); if (!response.ok) { throw new Error("Market feed unavailable"); } const result = await response.json(); const price = Number(result.data.p); if (!Number.isFinite(price)) { throw new Error("Invalid price received"); } const rows = document.querySelectorAll( "#dashboard .card table tbody tr" ); const eurusdRow = Array.from(rows).find( row => row.cells[0]?.textContent.trim() === "EURUSD" ); if (eurusdRow) { eurusdRow.cells[1].textContent = price.toFixed(5); eurusdRow.cells[2].textContent = "—"; eurusdRow.cells[2].className = "muted"; } const demoLabel = document.querySelector( ".topactions .pill:last-child" ); if (demoLabel) { demoLabel.textContent = "EURUSD feed connected"; demoLabel.classList.add("green"); } console.log("PipNex EURUSD price:", price); } catch (error) { console.error("EURUSD feed error:", error); } } loadEURUSD(); setInterval(loadEURUSD, 60000);
